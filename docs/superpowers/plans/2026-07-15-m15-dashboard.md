```php
<?php

namespace App\Http\Controllers;

use App\Models\Transversal\FirmaEvento;
use App\Models\User;
use App\Models\VentanillaUnica\Pqrs;
use App\Models\VentanillaUnica\Recibidos\VentanillaRadicaReci;
use App\Models\VentanillaUnica\Enviados\VentanillaRadicaEnviados;
use App\Models\OfiArchivo\Expediente;
use App\Models\OfiArchivo\Prestamo;
use App\Models\OfiArchivo\Transferencia;
use App\Services\OfiArchivo\ReportesService as ArchivoReportesService;
use App\Services\VentanillaUnica\PqrsService;
use App\Services\VentanillaUnica\Recibidos\VentanillaRadicaReciService;
use App\Services\VentanillaUnica\Enviados\VentanillaRadicaEnviadosService;
use App\Services\ControlAcceso\UserService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;

class DashboardGlobalController extends Controller
{
    public function __construct(
        protected ArchivoReportesService $archivoReportes,
        protected PqrsService $pqrsService,
        protected VentanillaRadicaReciService $reciService,
        protected VentanillaRadicaEnviadosService $enviadosService,
        protected UserService $userService,
    ) {}

    public function index(): \Illuminate\Http\JsonResponse
    {
        $data = Cache::remember('dashboard_global_stats', 300, function () {
            return $this->aggregateAll();
        });

        return response()->json(['success' => true, 'data' => $data]);
    }

    protected function aggregateAll(): array
    {
        $archivo = $this->archivoReportes->estadisticasGenerales();

        return [
            'radicados_recibidos' => $this->getEstadisticas('ventanilla_recibidos_estadisticas', fn() =>
                rescue(fn() => $this->reciService->estadisticas(), ['total' => 0, 'hoy' => 0, 'mes' => 0, 'por_estado' => []])
            ),
            'radicados_enviados' => $this->getEstadisticas('ventanilla_enviados_estadisticas', fn() =>
                rescue(fn() => $this->enviadosService->estadisticas(), ['total' => 0, 'hoy' => 0, 'mes' => 0, 'por_estado' => []])
            ),
            'pqrs' => $this->getEstadisticas('ventanilla_pqrs_estadisticas', fn() =>
                rescue(fn() => $this->pqrsService->estadisticas(), ['total' => 0, 'pendientes' => 0, 'urgentes' => 0, 'vencidas' => 0])
            ),
            'archivo' => [
                'expedientes' => $archivo['expedientes'] ?? ['total' => 0, 'abiertos' => 0, 'cerrados' => 0],
                'prestamos' => $archivo['prestamos'] ?? ['activos' => 0, 'vencidos' => 0],
                'transferencias' => $archivo['transferencias'] ?? ['pendientes' => 0, 'completadas_hoy' => 0],
                'sparklines' => $archivo['sparklines'] ?? [],
            ],
            'usuarios' => $this->getEstadisticas('users_statistics', fn() =>
                rescue(fn() => $this->userService->statistics(), ['total_users' => 0, 'activos' => 0, 'inactivos' => 0])
            ),
            'firmas' => $this->getFirmasStats(),
            'vencimientos_proximos' => $this->getVencimientosProximos(),
        ];
    }

    protected function getEstadisticas(string $cacheKey, \Closure $callback): array
    {
        return Cache::remember("dashboard_{$cacheKey}", 300, $callback);
    }

    protected function getFirmasStats(): array
    {
        return [
            'hoy' => FirmaEvento::whereDate('created_at', today())->count(),
            'semana' => FirmaEvento::where('created_at', '>=', now()->subWeek())->count(),
            'mes' => FirmaEvento::where('created_at', '>=', now()->subMonth())->count(),
        ];
    }

    protected function getVencimientosProximos(): array
    {
        $limite = now()->addDays(7);
        $items = [];

        // Radicados recibidos con fecha_vencimiento próxima
        $recibidos = VentanillaRadicaReci::where('fecha_vencimiento', '<=', $limite)
            ->where('fecha_vencimiento', '>=', now())
            ->where('estado', '!=', 'Finalizado')
            ->get(['id', 'radicado', 'asunto', 'fecha_vencimiento']);

        foreach ($recibidos as $r) {
            $items[] = [
                'tipo' => 'Radicado Recibido',
                'referencia' => $r->radicado,
                'descripcion' => Str::limit($r->asunto, 80),
                'fecha_vencimiento' => $r->fecha_vencimiento?->format('Y-m-d'),
                'dias_restantes' => now()->diffInDays($r->fecha_vencimiento, false),
            ];
        }

        // PQRS con fecha_vencimiento próxima
        $pqrs = Pqrs::where('fecha_vencimiento', '<=', $limite)
            ->where('fecha_vencimiento', '>=', now())
            ->whereNotIn('estado', ['Respondido', 'Cerrado'])
            ->get(['id', 'radicado', 'solicitante_nombre as asunto', 'fecha_vencimiento']);

        foreach ($pqrs as $p) {
            $items[] = [
                'tipo' => 'PQRS',
                'referencia' => $p->radicado,
                'descripcion' => 'PQRS: ' . ($p->asunto ?? ''),
                'fecha_vencimiento' => $p->fecha_vencimiento?->format('Y-m-d'),
                'dias_restantes' => now()->diffInDays($p->fecha_vencimiento, false),
            ];
        }

        // Préstamos vencidos
        $prestamos = Prestamo::where('fecha_devolucion', '<', now())
            ->where('estado', 'Activo')
            ->get(['id', 'solicitante_nombre as asunto', 'fecha_solicitud', 'fecha_devolucion']);

        foreach ($prestamos as $pr) {
            $items[] = [
                'tipo' => 'Préstamo Vencido',
                'referencia' => "P-{$pr->id}",
                'descripcion' => $pr->asunto ?? '',
                'fecha_vencimiento' => $pr->fecha_devolucion?->format('Y-m-d'),
                'dias_restantes' => now()->diffInDays($pr->fecha_devolucion, false),
            ];
        }

        usort($items, fn($a, $b) => ($a['dias_restantes'] ?? 999) <=> ($b['dias_restantes'] ?? 999));

        return $items;
    }
}
```

```js
'use client'

import { useCallback } from 'react'
import { useDashboardGlobalQuery } from '@/services/dashboard'

export const useDashboardGlobalHandlers = () => {
  const query = useDashboardGlobalQuery()

  const stats = query.data
  const isLoading = query.isLoading
  const isError = query.isError
  const error = query.error

  const handleRefetch = useCallback(() => {
    query.refetch()
  }, [query])

  const getModuleSemaphore = useCallback((modulo) => {
    if (!stats) return 'neutral'

    switch (modulo) {
      case 'recibidos': {
        const vencidos = stats.radicados_recibidos?.por_estado?.Vencido || 0
        const total = stats.radicados_recibidos?.total || 1
        return vencidos / total > 0.1 ? 'red' : vencidos > 0 ? 'yellow' : 'green'
      }
      case 'pqrs': {
        const urgentes = stats.pqrs?.urgentes || 0
        const vencidas = stats.pqrs?.vencidas || 0
        return urgentes > 0 || vencidas > 0 ? 'red' : 'green'
      }
      case 'archivo': {
        const vencidos = stats.archivo?.prestamos?.vencidos || 0
        return vencidos > 0 ? 'red' : 'green'
      }
      case 'usuarios': {
        return 'green'
      }
      default:
        return 'neutral'
    }
  }, [stats])

  return {
    stats,
    isLoading,
    isError,
    error,
    handleRefetch,
    getModuleSemaphore,
  }
}
```

```jsx
'use client'

import { useState } from 'react'
import dynamic from 'next/dynamic'
import { Box, Typography, Alert, CircularProgress, Card, CardContent } from '@mui/material'
import Grid from '@mui/material/Grid2'
import { useDashboardGlobalHandlers } from '@/hooks/dashboard'

const DashboardGlobalView = ({ dictionary = {} }) => {
  const { stats, isLoading, isError, getModuleSemaphore } = useDashboardGlobalHandlers()
  const [expandedModule, setExpandedModule] = useState(null)

  if (isLoading) {
    return (
      <Box sx={{ display: 'flex', justifyContent: 'center', py: 8 }}>
        <CircularProgress />
      </Box>
    )
  }

  if (isError) {

  }

  if (!stats) return null

  const modulos = [
    {
```

```jsx
import { Suspense } from 'react'
import { Breadcrumbs, Typography, Box } from '@mui/material'

import { getDictionary } from '@/utils/getDictionary'
import Link from '@/components/Link'
import DashboardGlobalClient from '@/views/dashboard/DashboardGlobalClient'

export const revalidate = 60

function DashboardLoading() {
  return (
    <Box sx={{ display: 'flex', justifyContent: 'center', py: 8 }}>

    const params = await props.params

              {dictionary['navigation_dashboard']?.global || 'Dashboard Global'}

          </Grid>
          <Grid size={{ xs: 12 }}>
            <DashboardGlobalClient dictionary={dictionary} locale={params.lang} />
          </Grid>
```
